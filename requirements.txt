# ============================================================================
# Robô da Lara — dependências
# Testado em Raspberry Pi OS Lite 64-bit (Bookworm/Trixie), Python 3.11 a 3.14
#
# Instalar (o venv TEM de ver os pacotes do sistema — é por lá que vem a câmara):
#     python3 -m venv --system-site-packages .venv && source .venv/bin/activate
#     pip install -r requirements.txt
#     pip install --no-deps "openwakeword>=0.6.0"      # ← ver "Voz: ouvir"
#
# No Raspberry Pi, ANTES disto, pelo apt (nunca pelo pip):
#     sudo apt install -y python3-lgpio python3-gpiozero
#     sudo apt install -y --no-install-recommends python3-picamera2
# ============================================================================

# --- Base -------------------------------------------------------------------
PyYAML>=6.0
numpy>=1.26,<3
requests>=2.32

# --- Visão ------------------------------------------------------------------
# YuNet (deteção) + SFace (reconhecimento) vêm DENTRO do OpenCV.
# Não é preciso instalar face_recognition nem compilar dlib.
# <5 de propósito: o OpenCV 5 saiu em 2026 e o código foi escrito para o 4.x.
#
# ⚠️ HEADLESS, e não o opencv-python normal. O pacote normal traz a parte
#    gráfica (janelas), que precisa do libGL.so.1 e do libglib — e o Raspberry
#    Pi OS **Lite** não tem nenhum dos dois. Instala bem e rebenta no import.
#    Este robô nunca abre uma janela: procurámos, e não há um único imshow,
#    namedWindow nem waitKey no projeto. O ver_camera.py mostra a imagem por
#    HTTP, não por janela. Portanto a versão gráfica seria só peso e uma
#    armadilha à espera.
opencv-python-headless>=4.10,<5

# --- Câmara (só no Raspberry Pi) --------------------------------------------
# picamera2 instala-se SÓ pelo apt (python3-picamera2). Um venv normal não o
# vê — daí o --system-site-packages lá em cima.
# picamera2

# --- Voz: falar -------------------------------------------------------------
# A síntese vem do mini. O Pi só reproduz o WAV e mantém frases em cache.

# --- Voz: ouvir -------------------------------------------------------------
# ⚠️ NÃO há Whisper no Pi. Quem transcreve é o mac mini — no Pi 5 o modelo que
#    percebe bem português fica mais lento que o tempo real, e são 500 MB no
#    cartão SD para um caso que a rede Tailscale torna raro. Sem mini, o robô
#    não percebe novas ordens faladas. Continua a reconhecer pessoas no Pi e
#    pode tocar avisos já guardados. O watchdog independente entra na etapa 3.
sounddevice>=0.4.7

# O áudio vai para o mini em contínuo, pelo WebSocket, enquanto a Lara fala.
# O `websocket-client` é síncrono, que é como o resto do robô está escrito.
websocket-client>=1.7

# ⚠️ openwakeword NÃO está nesta lista, de propósito.
# A versão 0.6.0 declara "tflite-runtime" como dependência obrigatória no
# Linux, e o tflite-runtime morreu no Python 3.11 (última versão: 2.14.0, de
# novembro de 2023). Com Python 3.12+ o pip rebenta em TODA a lista por causa
# dele. Nós só usamos o backend ONNX (wakeword.py: inference_framework="onnx"),
# por isso instala-se sem as dependências dele:
#     pip install --no-deps "openwakeword>=0.6.0"
# e as dependências que interessam mesmo vêm aqui, explicitamente:
onnxruntime>=1.20
scipy>=1.10
scikit-learn>=1.3
tqdm>=4.60

# --- Hardware ---------------------------------------------------------------
# gpiozero + lgpio: a forma CORRETA de usar GPIO no Pi 5.
# ⚠️ NUNCA instalar RPi.GPIO — não funciona no Pi 5 (chip RP1 novo).
# O lgpio vem do apt (python3-lgpio): o do PyPI compila C e falha no Mac.
# O gpiozero é puro Python; instalar pelo pip não faz mal em lado nenhum.
gpiozero>=2.0

# O mBot2 fica intacto. A biblioteca fala com o CyberPi e o shield por USB.
makeblock==0.1.8

# Porta série para o mBot2 e para o protótipo HUB75, se esse ecrã for escolhido.
pyserial>=3.5

# I2C para módulos adicionais, como servos ou monitorização de bateria.
# Falamos com eles diretamente por smbus2: sem Blinka, sem CircuitPython,
# sem nada que possa partir com a próxima atualização do Raspberry Pi OS.
smbus2>=0.5

# --- Desenvolvimento --------------------------------------------------------
pytest>=8.0
